[meta]
name        = "example-alpine-shell"
description = "Reference .tif showing every currently-modeled control/measure knob"

# ------------------------------------------------------------
# WORKLOAD — what actually runs
# ------------------------------------------------------------
[workload]
type    = "container"          # "container" | "model"
image   = "alpine:3.19"        # resolved to a rootfs under /var/lib/tif/images/
command = ["/bin/sh"]
workdir = "/"
env     = ["TERM=xterm", "LC_ALL=C"]

# ------------------------------------------------------------
# NAMESPACES — which isolation namespaces to unshare
# mechanism: clone(2) / unshare(2) flags
# ------------------------------------------------------------
[namespaces]
uts     = true   # CLONE_NEWUTS  — own hostname
pid     = true   # CLONE_NEWPID  — own PID tree, PID 1 inside
mount   = true   # CLONE_NEWNS   — own mount table
ipc     = true   # CLONE_NEWIPC  — own SysV IPC / POSIX mqueue namespace (not yet in your code — add when needed)
net     = false  # CLONE_NEWNET  — own network stack; false = share host netns (simplest, no veth plumbing yet)
user    = false  # CLONE_NEWUSER — UID/GID remapping; leave false until you implement uid_map/gid_map writing

# ------------------------------------------------------------
# ROOTFS / MOUNTS
# mechanism: chroot(2) or pivot_root(2), mount(2)
# ------------------------------------------------------------
[rootfs]
path      = "./rootfs"
readonly  = false        # remount root MS_RDONLY after setup
pivot     = false        # false = chroot (current impl); true = pivot_root (safer, escapes chroot-breakout class of bugs)

[[mounts]]               # extra mounts beyond the implicit /proc
source = "tmpfs"
target = "/dev/shm"
type   = "tmpfs"
options = ["nosuid", "nodev", "size=64m"]

# ------------------------------------------------------------
# RESOURCES — process-level rlimits
# mechanism: setrlimit(2), applied before exec in the child
# ------------------------------------------------------------
[resources]
memory_mb     = 512      # informational mirror of resources.memory.max_mb below; kept for quick reading
swap_mb       = 0
max_procs     = 100      # mirrors cgroup pids.max; rlimit RLIMIT_NPROC as defense-in-depth
nofile        = 1024     # RLIMIT_NOFILE
core_dump     = false    # RLIMIT_CORE = 0 if false

# ------------------------------------------------------------
# CPU
# mechanism: cgroup v2 cpu.max / cpu.weight / cpuset.cpus / cpuset.mems,
#            plus per-core cpufreq sysfs writes (host-wide, not per-cgroup)
# ------------------------------------------------------------
[cpu]
cores       = 4                  # informational: number of logical CPUs the workload expects
affinity    = [0, 1, 2, 3]       # -> cpuset.cpus, hard pin (kernel will not schedule outside this set)
core_type   = "p"                # "p" | "e" | "any" — resolved to concrete core IDs via
                                  #   /sys/devices/system/cpu/cpu*/topology/core_type on hybrid CPUs;
                                  #   falls back to a static topology table if that file is absent
numa_nodes  = []                 # -> cpuset.mems; leave empty on non-NUMA hosts (out of scope per paper1)
weight      = 100                # -> cpu.weight (1-10000), relative share under contention, default 100 = neutral
quota_us    = 400000             # -> cpu.max first field: microseconds of CPU time allowed per period
period_us   = 100000             # -> cpu.max second field: period length; quota/period = 4.0 = 4 cores worth
governor    = "performance"      # per-core scaling_governor for cores in `affinity`; "performance" disables
                                  # frequency scaling down, reduces run-to-run latency variance
turbo       = false              # writes /sys/devices/system/cpu/intel_pstate/no_turbo = 1 (or amd_pstate equiv);
                                  # false here means "turbo disabled" for reproducible benchmarking
smt         = "unchanged"        # "unchanged" | "off" — host-wide SMT/HT toggle, use with extreme caution,
                                  # affects cores outside this workload too; leave "unchanged" unless benchmarking

# ------------------------------------------------------------
# SCHEDULING — per-thread policy, finer-grained than cgroup weight
# mechanism: sched_setattr(2) / sched_setscheduler(2)
# ------------------------------------------------------------
[scheduling]
policy   = "other"      # "other" (CFS/EEVDF default) | "fifo" | "rr" | "deadline" — leave "other" unless
                         # you have a specific real-time reason; fifo/rr require root and can starve the host
priority = 0             # nice value equivalent, -20..19, only meaningful when policy = "other"

# ------------------------------------------------------------
# MEMORY
# mechanism: cgroup v2 memory.max / memory.high / memory.swap.max,
#            mlockall(2), prctl(PR_SET_THP_DISABLE)
# ------------------------------------------------------------
[memory]
max_mb   = 512            # -> memory.max, hard limit, OOM-kills the cgroup if exceeded
high_mb  = 448             # -> memory.high, soft throttle point below max; kernel reclaims aggressively
                            # instead of killing — set ~85-90% of max_mb for graceful degradation
swap_mb  = 0                # -> memory.swap.max
mlock    = false             # mlockall(MCL_CURRENT|MCL_FUTURE) — pins all pages resident, no swap-induced
                              # latency spikes; costs you the memory up front, only worth it for latency-critical
thp      = "unchanged"        # "unchanged" | "never" | "madvise" — prctl(PR_SET_THP_DISABLE) if "never"

# ------------------------------------------------------------
# I/O
# mechanism: cgroup v2 io.max / io.weight (per block device, keyed by major:minor)
# ------------------------------------------------------------
[io]
weight = 100                     # -> io.weight, relative priority, 1-10000

  [[io.limits]]                  # -> io.max, one entry per device
  device = "8:0"                 # major:minor, find via `lsblk -o NAME,MAJ:MIN`
  rbps   = "500M"                # read bytes/sec cap
  wbps   = "100M"                # write bytes/sec cap
  riops  = 0                     # 0 = unset/unlimited
  wiops  = 0

# ------------------------------------------------------------
# CACHE / MEMORY BANDWIDTH
# mechanism: resctrl filesystem (Intel CAT/MBA, AMD equivalent) — HARDWARE-GATED,
#            not available on most consumer CPUs. Runtime checks for
#            /sys/fs/resctrl before applying; warns and no-ops if absent.
# ------------------------------------------------------------
[cache]
l3_partition = ""        # e.g. "0-1" — CLOS bitmask range for CAT; empty = don't touch resctrl at all
mba_percent  = 100        # Memory Bandwidth Allocation cap, 10-100 in steps of 10 on most Intel platforms

# ------------------------------------------------------------
# CAPABILITIES / SECURITY
# mechanism: capset(2), seccomp(2) via prctl(PR_SET_SECCOMP) or seccomp(2) syscall
# ------------------------------------------------------------
[security]
capabilities_drop = ["CAP_SYS_ADMIN", "CAP_NET_RAW", "CAP_SYS_MODULE"]   # bounding set removals
seccomp_profile    = "default"    # "default" | "unconfined" | path to a custom JSON profile

# ------------------------------------------------------------
# NETWORK
# ------------------------------------------------------------
[network]
mode = "none"      # "none" | "host" | "bridge" — bridge requires netns=true above + veth plumbing (future work)

# ------------------------------------------------------------
# TELEMETRY — general profiler (Profiler A)
# mechanism: /proc/<pid>/{stat,status,schedstat}, cgroup v2 stat files, PSI

# ------------------------------------------------------------
[telemetry]
enabled     = true
profiler    = "general"      # "general" | "profinfer" (profinfer only valid for workload.type = "model")
interval_ms = 1000
export      = "csv"           # "csv" | "sqlite"
output_dir  = "/var/lib/tif/telemetry/example-alpine-shell/"

  [telemetry.metrics]         # toggle individual cheap metrics on/off
  cpu_percent        = true    # derived from cpu.stat usage_usec delta / wall time
  rss                = true    # memory.current
  memory_peak        = true    # memory.peak (cgroup v2 exposes directly)
  memory_events       = true    # memory.events: low/high/max/oom/oom_kill counts
  pids_current        = true    # pids.current
  ctx_switches        = true    # /proc/<pid>/status: voluntary_ctxt_switches, nonvoluntary_ctxt_switches
  migrations          = true    # sampled from /proc/<pid>/stat field 39 (processor), diffed over time
  runqueue_latency    = true    # /proc/<pid>/schedstat field 2 — time spent waiting to be scheduled
  psi_cpu             = true    # /sys/fs/cgroup/<id>/cpu.pressure — "some avg10" / "full avg10"
  psi_memory          = true    # /sys/fs/cgroup/<id>/memory.pressure
  psi_io              = true    # /sys/fs/cgroup/<id>/io.pressure
  io_stat             = true    # io.stat — bytes/ops per device
  major_faults        = true    # /proc/<pid>/stat fields 12/13 (majflt, cmajflt)
